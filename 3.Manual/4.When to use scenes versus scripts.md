## ពេលណាដែលគួរប្រើស្គីន (Scenes) ធៀបនឹងស្គ្រីប (Scripts)

យើងបានគ្របដណ្តប់រួចហើយថាតើស្គីន (scenes) និងស្គ្រីប (scripts) ខុសគ្នាយ៉ាងដូចម្តេច។ ស្គ្រីបកំណត់ការពង្រីកថ្នាក់ (class extension) របស់អេនជីនដោយប្រើកូដអនុវត្ត (imperative code) ខណៈដែលស្គីនប្រើកូដប្រកាស (declarative code)។

សមត្ថភាពនៃប្រព័ន្ធនីមួយៗគឺខុសគ្នាជាលទ្ធផល។ ស្គីនអាចកំណត់ពីរបៀបដែលថ្នាក់ដែលបានពង្រីកចាប់ផ្តើម (initialize) ប៉ុន្តែមិនមែនជាអ្វីដែលជាអាកប្បកិរិយា (behavior) ជាក់ស្តែងរបស់វានោះទេ។ ស្គីនត្រូវបានគេប្រើជាញឹកញាប់រួមជាមួយស្គ្រីបមួយ ដោយស្គីនប្រកាសពីការរួមបញ្ចូលគ្នានៃថ្នាំង (nodes) ហើយស្គ្រីបបន្ថែមអាកប្បកិរិយាដោយប្រើកូដអនុវត្ត។

## ប្រភេទគ្មានឈ្មោះ (Anonymous types)[](https://docs.godotengine.org/en/stable/tutorials/best_practices/scenes_versus_scripts.html#anonymous-types "Link to this heading")

វា *អាច* ធ្វើទៅបានក្នុងការកំណត់ខ្លឹមសាររបស់ស្គីនទាំងស្រុងដោយប្រើតែស្គ្រីបមួយ។ នេះគឺជាមូលដ្ឋានគ្រឹះនៃអ្វីដែលអេឌីទ័រ Godot ធ្វើ គឺត្រឹមតែនៅក្នុងកុងស្ត្រុកទ័រ (constructor) C++ របស់វត្ថុរបស់វាប៉ុណ្ណោះ។

ប៉ុន្តែ ការជ្រើសរើសថាតើត្រូវប្រើមួយណាអាចជាបញ្ហាលំបាក។ ការបង្កើតឧទាហរណ៍ស្គ្រីប (script instances) គឺដូចគ្នានឹងការបង្កើតថ្នាក់នៅក្នុងអេនជីន ខណៈដែលការដោះស្រាយស្គីនត្រូវការការផ្លាស់ប្តូរនៅក្នុង API៖

```
const MyNode = preload("my_node.gd")
const MyScene = preload("my_scene.tscn")
var node = Node.new()
var my_node = MyNode.new() # Same method call.
var my_scene = MyScene.instantiate() # Different method call.
var my_inherited_scene = MyScene.instantiate(PackedScene.GEN_EDIT_STATE_MAIN) # Create scene inheriting from MyScene.
```

ផងដែរ ស្គ្រីបនឹងដំណើរការយឺតជាងស្គីនបន្តិច ដោយសារភាពខុសគ្នានៃល្បឿនរវាងកូដអេនជីន និងកូដស្គ្រីប។ ថ្នាំង (node) កាន់តែធំ និងកាន់តែស្មុគស្មាញ កាន់តែមានមូលហេតុច្រើនដើម្បីសាងសង់វាជាស្គីន។

## ប្រភេទមានឈ្មោះ (Named types)[](https://docs.godotengine.org/en/stable/tutorials/best_practices/scenes_versus_scripts.html#named-types "Link to this heading")

ស្គ្រីបអាចត្រូវបានចុះឈ្មោះជាប្រភេទថ្មីនៅក្នុងអេឌីទ័រខ្លួនឯង។ នេះបង្ហាញវាជាប្រភេទថ្មីនៅក្នុងប្រអប់បង្កើតថ្នាំង ឬធនធាន (node or resource creation dialog) ជាមួយនឹងរូបតំណាង (icon) ជាជម្រើស។ តាមរបៀបនេះ សមត្ថភាពរបស់អ្នកប្រើប្រាស់ក្នុងការប្រើប្រាស់ស្គ្រីបគឺមានភាពរលូនជាងមុន។ ជំនួសឱ្យការត្រូវ...

1.  ដឹងពីប្រភេទមូលដ្ឋាន (base type) នៃស្គ្រីបដែលពួកគេចង់ប្រើ។
    
2.  បង្កើតឧទាហរណ៍នៃប្រភេទមូលដ្ឋាននោះ។
    
3.  បន្ថែមស្គ្រីបទៅលើថ្នាំង។
    

ជាមួយស្គ្រីបដែលបានចុះឈ្មោះ ប្រភេទស្គ្រីបជំនួសមកវិញក្លាយជាជម្រើសនៃការបង្កើតដូចជាថ្នាំង និងធនធានផ្សេងទៀតនៅក្នុងប្រព័ន្ធ។ ប្រអប់បង្កើត (creation dialog) ថែមទាំងមានរបារស្វែងរក (search bar) ដើម្បីស្វែងរកប្រភេទតាមឈ្មោះ។

មានប្រព័ន្ធពីរសម្រាប់ការចុះឈ្មោះប្រភេទ៖

-   [ប្រភេទផ្ទាល់ខ្លួន (Custom Types)](https://docs.godotengine.org/en/stable/tutorials/plugins/editor/making_plugins.html#doc-making-plugins)
    
    > -   សម្រាប់អេឌីទ័រប៉ុណ្ណោះ។ ឈ្មោះប្រភេទ (Typenames) មិនអាចចូលប្រើបាននៅពេលដំណើរការ (runtime)។
    >     
    > -   មិនគាំទ្រប្រភេទផ្ទាល់ខ្លួនដែលមានការតមកេរ (inherited custom types)។
    >     
    > -   ឧបករណ៍ចាប់ផ្តើម (initializer tool)។ បង្កើតថ្នាំងជាមួយស្គ្រីប។ គ្មានអ្វីច្រើនជាងនេះទេ។
    >     
    > -   អេឌីទ័រមិនមានការយល់ដឹងអំពីប្រភេទ (type-awareness) ចំពោះស្គ្រីប ឬទំនាក់ទំនងរបស់វាជាមួយប្រភេទ ឬស្គ្រីបផ្សេងទៀតរបស់អេនជីននោះទេ។
    >     
    > -   អនុញ្ញាតឱ្យអ្នកប្រើប្រាស់កំណត់រូបតំណាង (icon)។
    >     
    > -   ដំណើរការសម្រាប់ភាសាស្គ្រីបទាំងអស់ ព្រោះវាដោះស្រាយជាមួយធនធានស្គ្រីប (Script resources) ជាលក្ខណៈសកល។
    >     
    > -   រៀបចំដោយប្រើ [EditorPlugin.add_custom_type](https://docs.godotengine.org/en/stable/classes/class_editorplugin.html#class-editorplugin-method-add-custom-type)។
    >     
    
-   [ថ្នាក់ស្គ្រីប (Script Classes)](https://docs.godotengine.org/en/stable/tutorials/scripting/gdscript/gdscript_basics.html#doc-gdscript-basics-class-name)
    
    > -   អាចចូលប្រើបានទាំងនៅក្នុងអេឌីទ័រ និងពេលដំណើរការ (Editor and runtime accessible)។
    >     
    > -   បង្ហាញទំនាក់ទំនងតមកេរ (inheritance relationships) ពេញលេញ។
    >     
    > -   បង្កើតថ្នាំងជាមួយស្គ្រីប ប៉ុន្តែក៏អាចផ្លាស់ប្តូរប្រភេទ ឬពង្រីកប្រភេទពីអេឌីទ័រផងដែរ។
    >     
    > -   អេឌីទ័រដឹងពីទំនាក់ទំនងតមកេររវាងស្គ្រីប ថ្នាក់ស្គ្រីប និងថ្នាក់ C++ របស់អេនជីន។
    >     
    > -   អនុញ្ញាតឱ្យអ្នកប្រើប្រាស់កំណត់រូបតំណាង (icon)។
    >     
    > -   អ្នកអភិវឌ្ឍន៍អេនជីនត្រូវបន្ថែមការគាំទ្រសម្រាប់ភាសាដោយដៃ (ការបង្ហាញឈ្មោះ និងការចូលប្រើពេលដំណើរការ)។
    >     
    > -   អេឌីទ័រស្កេនថតឯកសារគម្រោង (project folders) និងចុះឈ្មោះឈ្មោះដែលបានបង្ហាញណាមួយសម្រាប់ភាសាស្គ្រីបទាំងអស់។ ភាសាស្គ្រីបនីមួយៗត្រូវតែអនុវត្តការគាំទ្រផ្ទាល់ខ្លួនរបស់វាសម្រាប់ការបង្ហាញព័ត៌មាននេះ។
    >     
    
វិធីសាស្ត្រទាំងពីរបន្ថែមឈ្មោះទៅក្នុងប្រអប់បង្កើត (creation dialog) ប៉ុន្តែជាពិសេសថ្នាក់ស្គ្រីប (script classes) ក៏អនុញ្ញាតឱ្យអ្នកប្រើប្រាស់ចូលប្រើឈ្មោះប្រភេទ (typename) ដោយមិនចាំបាច់ផ្ទុកធនធានស្គ្រីបផងដែរ។ ការបង្កើតឧទាហរណ៍ និងការចូលប្រើថេរ (constants) ឬមេតូដស្តាទិច (static methods) គឺអាចធ្វើទៅបានពីគ្រប់ទីកន្លែង។

ជាមួយនឹងមុខងារបែបនេះ អ្នកម្នាក់អាចចង់ឱ្យប្រភេទរបស់ពួកគេជាស្គ្រីបដោយគ្មានស្គីន ដោយសារតែភាពងាយស្រួលនៃការប្រើប្រាស់ដែលវាផ្តល់ជូនអ្នកប្រើប្រាស់។ អ្នកដែលកំពុងអភិវឌ្ឍន៍កម្មវិធីជំនួយ (plugins) ឬបង្កើតឧបករណ៍ផ្ទៃក្នុងសម្រាប់អ្នករចនាប្រើប្រាស់ នឹងរកឃើញភាពងាយស្រួលក្នុងការធ្វើការតាមរបៀបនេះ។

នៅផ្នែកអវិជ្ជមាន វាក៏មានន័យថាត្រូវប្រើការសរសេរកម្មវិធីបែបអនុវត្ត (imperative programming) ភាគច្រើនផងដែរ។

## ដំណើរការរបស់ស្គ្រីប ធៀបនឹង PackedScene[](https://docs.godotengine.org/en/stable/tutorials/best_practices/scenes_versus_scripts.html#performance-of-script-vs-packedscene "Link to this heading")

ផ្នែកចុងក្រោយមួយទៀតដែលត្រូវពិចារណានៅពេលជ្រើសរើសស្គីន និងស្គ្រីបគឺល្បឿននៃការអនុវត្ត (execution speed)។

នៅពេលដែលទំហំនៃវត្ថុកើនឡើង ទំហំចាំបាច់នៃស្គ្រីបដើម្បីបង្កើត និងចាប់ផ្តើម (initialize) ពួកវាក៏កើនឡើងធំជាងមុនដែរ។ ការបង្កើតថ្នាំងជាន់ (node hierarchies) បង្ហាញពីរឿងនេះ។ តក្កវិជ្ជា (logic) របស់ថ្នាំង (Node) នីមួយៗអាចមានប្រវែងកូដរាប់រយបន្ទាត់។

ឧទាហរណ៍កូដខាងក្រោមបង្កើត `Node` ថ្មីមួយ ផ្លាស់ប្តូរឈ្មោះរបស់វា កំណត់ស្គ្រីបមួយឱ្យវា កំណត់មេរបស់វានាពេលអនាគត (future parent) ជាម្ចាស់ (owner) របស់វា ដើម្បីឱ្យវាត្រូវបានរក្សាទុកទៅក្នុងថាស (disk) ជាមួយវា ហើយទីបំផុតបន្ថែមវាជាកូន (child) របស់ថ្នាំង `Main`៖

```
# main.gd
extends Node

func _init():
	var child = Node.new()
	child.name = "Child"
	child.script = preload("child.gd")
	add_child(child)
	child.owner = self
```

កូដស្គ្រីបដូចនេះគឺយឺតជាងកូដ C++ ផ្នែកអេនជីន (engine-side) ច្រើន។ ការណែនាំ (instruction) នីមួយៗធ្វើការហៅទៅកាន់ API ស្គ្រីប (scripting API) ដែលនាំឱ្យមានការ "ស្វែងរក" (lookups) ជាច្រើននៅផ្នែកខាងក្រោយ (back-end) ដើម្បីរកតក្កវិជ្ជាដែលត្រូវអនុវត្ត។

ស្គីនជួយជៀសវាងបញ្ហាដំណើរការនេះ។ [PackedScene](https://docs.godotengine.org/en/stable/classes/class_packedscene.html#class-packedscene) ដែលជាប្រភេទមូលដ្ឋានដែលស្គីនទទួលមក (inherit from) កំណត់ធនធានដែលប្រើទិន្នន័យស៊េរី (serialized data) ដើម្បីបង្កើតវត្ថុ។ អេនជីនអាចដំណើរការស្គីនជាក្រុម (batches) នៅផ្នែកខាងក្រោយ (back-end) និងផ្តល់នូវដំណើរការដែលល្អប្រសើរជាងស្គ្រីបច្រើន។

## សេចក្តីសន្និដ្ឋាន[](https://docs.godotengine.org/en/stable/tutorials/best_practices/scenes_versus_scripts.html#conclusion "Link to this heading")

នៅទីបំផុត វិធីសាស្ត្រល្អបំផុតគឺត្រូវពិចារណាលើចំណុចដូចខាងក្រោម៖

-   ប្រសិនបើអ្នកណាម្នាក់ចង់បង្កើតឧបករណ៍មូលដ្ឋាន (basic tool) ដែលនឹងត្រូវបានប្រើប្រាស់ឡើងវិញនៅក្នុងគម្រោងផ្សេងៗគ្នាជាច្រើន ហើយដែលមនុស្សគ្រប់កម្រិតជំនាញទំនងជានឹងប្រើ (រួមទាំងអ្នកដែលមិនចាត់ទុកខ្លួនឯងថាជា "អ្នកសរសេរកម្មវិធី") នោះទំនងជាវាគួរតែជាស្គ្រីប ទំនងជាមួយឈ្មោះ/រូបតំណាងផ្ទាល់ខ្លួន (custom name/icon)។
    
-   ប្រសិនបើអ្នកណាម្នាក់ចង់បង្កើតគំនិត (concept) ដែលជាក់លាក់ចំពោះហ្គេមរបស់ពួកគេ នោះវាគួរតែជាស្គីនជានិច្ច។ ស្គីនគឺងាយស្រួលក្នុងការតាមដាន/កែសម្រួល និងផ្តល់នូវសុវត្ថិភាពច្រើនជាងស្គ្រីប។
    
-   ប្រសិនបើអ្នកណាម្នាក់ចង់ដាក់ឈ្មោះឱ្យស្គីនមួយ នោះពួកគេនៅតែអាចធ្វើបែបនេះបានដោយការប្រកាសថ្នាក់ស្គ្រីប (script class) ហើយផ្តល់ឱ្យវានូវស្គីនមួយជាថេរ (constant)។ ស្គ្រីបក្លាយជា ជាប្រសិទ្ធភាព ជាចន្លោះឈ្មោះ (namespace)៖
    
    ```
    # game.gd
    class_name Game # extends RefCounted, so it won't show up in the node creation dialog.
    extends RefCounted
    
    const MyScene = preload("my_scene.tscn")
    
    # main.gd
    extends Node
    func _ready():
    	add_child(Game.MyScene.instantiate())
    ```

> **កំណត់សម្គាល់៖** ទំព័រនេះមាន 1 iframe ដែលមកពីប្រភពខុសគ្នា (cross-origin) ដែលមិនអាចចូលប្រើបានដោយសារតែគោលនយោបាយសុវត្ថិភាពរបស់កម្មវិធីរុករក (browser)។ ខ្លឹមសារមួយចំនួនអាចនឹងបាត់។ តំណភ្ជាប់ទៅកាន់ iframes ទាំងនេះត្រូវបានរក្សាទុកនៅកន្លែងដែលអាចធ្វើទៅបាន។
